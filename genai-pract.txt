from diffusers import StableDiffusionPipeline
import torch

pipe = StableDiffusionPipeline.from_pretrained(
    "runwayml/stable-diffusion-v1-5",
    torch_dtype = torch.float16
).to("cuda")


prompt = input("Enter prompt: ")


image = pipe(prompt).images[0]


image.save("output.png")
image.show()

!pip install -q transformers torch pillow

from transformers import BlipProcessor, BlipForConditionalGeneration
from PIL import Image

processor = BlipProcessor.from_pretrained(
    "Salesforce/blip-image-captioning-base"
)

model = BlipForConditionalGeneration.from_pretrained(
    "Salesforce/blip-image-captioning-base"
)

image = Image.open("image.jpg")

inputs = processor(images=image, return_tensors="pt")
output = model.generate(**inputs)

text = processor.decode(output[0], skip_special_tokens=True)

print("Generated Text:", text)

from transformers import pipeline

chatbot = pipeline("text-generation", model="microsoft/DialoGPT-small")

while True:
    user = input("You: ")

    if user.lower() == "exit":
        print("Bot: Goodbye!")
        break

    response = chatbot(
        user,
        max_new_tokens=30,
        pad_token_id=50256
    )

-----------------------------------Image to Sketch Generation------------------------------------------------------------------------
!pip install -q diffusers transformers accelerate controlnet_aux

import cv2, torch
from PIL import Image
from google.colab import files
from diffusers import ControlNetModel, StableDiffusionControlNetPipeline
from controlnet_aux import CannyDetector

# Upload
f = files.upload()
img = cv2.imread(next(iter(f)))

# Canny edges
edges = CannyDetector()(img)
edges = Image.fromarray(edges).convert("RGB").resize((512, 512))

# Models
controlnet = ControlNetModel.from_pretrained(
    "lllyasviel/sd-controlnet-canny",
    torch_dtype=torch.float16
)

pipe = StableDiffusionControlNetPipeline.from_pretrained(
    "runwayml/stable-diffusion-v1-5",
    controlnet=controlnet,
    torch_dtype=torch.float16
).to("cuda")

# Generate
sketch = pipe(
    "black and white pencil sketch",
    image=edges,
    num_inference_steps=20
).images[0]

sketch.save("sketch.png")
display(sketch)
